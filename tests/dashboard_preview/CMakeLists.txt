cmake_minimum_required(VERSION 3.16)
project(toucan_dashboard_preview C)
set(LV_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE STRING "" FORCE)
set(LV_CONF_INCLUDE_SIMPLE ON CACHE BOOL "" FORCE)
set(LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
file(GLOB_RECURSE LVGL_FILES "${LVGL_SOURCE}/src/*.c")
add_library(lvgl STATIC ${LVGL_FILES})
target_include_directories(lvgl PUBLIC ${LVGL_SOURCE} ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_definitions(lvgl PUBLIC LV_CONF_INCLUDE_SIMPLE=1)
set(STUBS "zephyr/kernel.h;zmk/display.h;zmk/endpoints.h;zmk/hid.h;zmk/keymap.h;zmk/event_manager.h;zmk/events/battery_state_changed.h;zmk/events/split_central_status_changed.h;braille_display.h;braille_engine.h;braille_tables.h;braille_editor.h;braille_storage.h")
foreach(header IN LISTS STUBS)
 get_filename_component(dir "${CMAKE_BINARY_DIR}/stubs/${header}" DIRECTORY)
 file(MAKE_DIRECTORY "${dir}")
 file(WRITE "${CMAKE_BINARY_DIR}/stubs/${header}" "#include \"platform.h\"\n")
endforeach()
add_executable(dashboard-preview main.c)
target_include_directories(dashboard-preview PRIVATE ${CMAKE_CURRENT_SOURCE_DIR} ${CMAKE_BINARY_DIR}/stubs)
target_link_libraries(dashboard-preview PRIVATE lvgl m)
